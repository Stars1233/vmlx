    78	 * UI already stores the native calls/results, per-generation reasoning

    88	  const results = validResults(message.toolResultsOaiJson)

    90	  const segments = persistedReasoningSegments(message)
    91

    93	  for (const status of parseArray(message.toolCallsJson)) {

    96	      typeof status.toolCallId === 'string' &&
