 14049	def _responses_output_to_assistant_messages(output_items: list) -> list[dict]:
 14050	    """Convert a Responses output array back to chat-history assistant turns."""
 14051	    assistant_messages: list[dict] = []
 14052	    content_parts: list[str] = []
 14053	    tool_calls: list[dict] = []
 14054	    reasoning_parts: list[str] = []
 14055
 14056	    for item in output_items or []:
 14057	        item_type = _responses_field(item, "type")
 14058	        if item_type in _RESPONSES_REASONING_ITEM_TYPES:
 14059	            reasoning_parts.extend(_responses_reasoning_texts(item))
 14060	            continue
 14061	        if item_type == "function_call":
 14062	            name = _responses_field(item, "name", "")
 14063	            arguments = _responses_field(item, "arguments", "{}")
 14064	            call_id = _responses_field(item, "call_id") or f"call_{uuid.uuid4().hex[:8]}"
 14065	            tool_calls.append(
 14066	                {
 14067	                    "id": call_id,
 14068	                    "type": "function",
 14069	                    "function": {"name": name, "arguments": arguments},
 14070	                }
 14071	            )
 14072	            continue
 14073	        content = _responses_field(item, "content") or []
 14074	        for part in content:
 14075	            ptype = _responses_field(part, "type")
 14076	            text = _responses_part_text(part)
 14077	            if ptype in _RESPONSES_VISIBLE_PART_TYPES and text:
 14078	                content_parts.append(text)
 14079	            elif ptype in _RESPONSES_REASONING_PART_TYPES:
 14080	                if text:
 14081	                    reasoning_parts.append(text)

 14087	        # strict native chat templates.
 14088	        message = {
 14089	            "role": "assistant",
 14090	            "content": "\n".join(content_parts) if content_parts else "",
 14091	            "tool_calls": tool_calls,
 14092	        }

 14095	        assistant_messages.append(message)

 14101	    elif reasoning_text and not assistant_messages:
 14102	        # Reasoning-only output (no visible text, no tool calls): inject an
 14103	        # assistant turn so `previous_response_id` chains preserve both the
 14104	        # user→assistant→user anchor and the model's actual reasoning rail.
 14105	        assistant_messages.append(
 14106	            {
 14107	                "role": "assistant",
 14108	                "content": "",
 14109	                "reasoning_content": reasoning_text,
 14110	            }
 14111	        )
 14112	    return assistant_messages

 14115	def _responses_store_history(
 14116	    response_id: str,
 14117	    messages: list[dict],
 14118	    *,
 14119	    reasoning_only: bool = False,
 14120	) -> None:
 14121	    """Persist a Responses history slot keyed by response_id.
 14122
 14123	    Args:
 14124	        response_id: identifier returned in `ResponsesObject.id`.
 14125	        messages: the chat-history messages to store (input + replayed
 14126	            assistant turns from `_responses_output_to_assistant_messages`).
 14127	        reasoning_only: True when the response produced ONLY reasoning items
 14128	            (no visible text, no tool calls). Tracked in
 14129	            `_responses_was_reasoning_only` so chained turns can surface a
 14130	            warning to the client.
 14131	    """
 14132	    if not response_id:
 14133	        return
 14134	    with _responses_history_lock:
 14135	        _responses_history[response_id] = _clone_response_messages(messages)
 14136	        _responses_history.move_to_end(response_id)
 14137	        if reasoning_only:
 14138	            _responses_was_reasoning_only.add(response_id)
 14139	        else:
 14140	            _responses_was_reasoning_only.discard(response_id)
 14141	        # Capacity enforcement — also evict the matching reasoning-only
 14142	        # marker so the marker set never references a freed slot (would
 14143	        # otherwise give stale-warning false positives if a future
 14144	        # response_id collides via uuid prefix).
 14145	        while len(_responses_history) > _RESPONSES_HISTORY_MAX:
 14146	            evicted_id, _ = _responses_history.popitem(last=False)
 14147	            _responses_was_reasoning_only.discard(evicted_id)
