   176	def _mllm_media_cache_extra_keys(request: Any) -> Optional[Dict[str, str]]:
   177	    """Return a stable media fingerprint for paged VLM prefix-cache keys.
   178
   179	    Token ids are not enough for media prompts: two different images with the
   180	    same grid shape render the same placeholder tokens. The model state after
   181	    prefill, however, depends on pixel values and media grids. This side key is
   182	    mixed into paged/block hashes while token counts stay based on real model
   183	    tokens only.
   184	    """
   185	    if request is None:
   186	        return None
   187	    has_media = bool(
   188	        getattr(request, "images", None)
   189	        or getattr(request, "videos", None)
   190	        or getattr(request, "audio_codes", None) is not None
   191	        or getattr(request, "audio_embeds", None) is not None
   192	        or getattr(request, "audio_features", None) is not None
   193	        or getattr(request, "audio_features_mask", None) is not None
   194	        or getattr(request, "audio", None)
   195	        or getattr(request, "audios", None)
   196	        or getattr(request, "pixel_values", None) is not None
   197	        or getattr(request, "image_grid_thw", None) is not None
   198	        or getattr(request, "pixel_values_videos", None) is not None
   199	        or getattr(request, "video_pixel_values", None) is not None
   200	        or getattr(request, "video_grid_thw", None) is not None
   201	    )
   202	    if not has_media:
   203	        return None
   204
   205	    hasher = hashlib.sha256()
   206	    hasher.update(b"vmlx-mllm-media-cache-v1")
   207	    media_sources = list(getattr(request, "images", None) or []) + list(
   208	        getattr(request, "videos", None) or []
   209	    )
   210
   211	    def _hash_text(label: str, value: Any) -> None:
   212	        hasher.update(label.encode("utf-8"))
   213	        hasher.update(hashlib.sha256(str(value).encode("utf-8")).digest())
   214
   215	    def _hash_media_source(label: str, value: Any) -> None:
   216	        """Hash local media by bytes so equivalent temp paths share a key."""
   217	        try:
   218	            path = Path(value)
   219	            if path.is_file():
   220	                content_hash = hashlib.sha256()
   221	                with path.open("rb") as handle:
   222	                    for chunk in iter(lambda: handle.read(1024 * 1024), b""):
   223	                        content_hash.update(chunk)
   224	                hasher.update(label.encode("utf-8"))
   225	                hasher.update(b"local-media-content")
   226	                hasher.update(content_hash.digest())
   227	                return
   228	        except (OSError, TypeError, ValueError):
   229	            pass
   230	        _hash_text(label, value)
   231
   232	    def _hash_array(label: str, value: Any) -> None:
   233	        if value is None:
   234	            return
   235	        hasher.update(label.encode("utf-8"))
   236	        try:
   237	            hasher.update(str(getattr(value, "shape", "")).encode("utf-8"))
   238	            hasher.update(str(getattr(value, "dtype", "")).encode("utf-8"))
   239	            import numpy as np
   240
   241	            hasher.update(np.array(value).tobytes())
   242	        except Exception:
   243	            _hash_text(label, value)
   244
   245	    for source in getattr(request, "images", None) or []:
   246	        _hash_media_source("image", source)
   247	    for source in getattr(request, "videos", None) or []:
   248	        _hash_media_source("video", source)
   249	    if getattr(request, "videos", None):
   250	        _hash_text("video_fps", getattr(request, "video_fps", None))
   251	        _hash_text("video_max_frames", getattr(request, "video_max_frames", None))
   252	    for source in getattr(request, "audios", None) or []:
   253	        _hash_media_source("audio", source)
   254	    if getattr(request, "audio", None):
   255	        _hash_text("audio", getattr(request, "audio", None))
   256	    _hash_array("image_grid_thw", getattr(request, "image_grid_thw", None))
   257	    _hash_array("video_grid_thw", getattr(request, "video_grid_thw", None))
   258	    _hash_array("audio_codes", getattr(request, "audio_codes", None))
   259	    _hash_array("audio_embeds", getattr(request, "audio_embeds", None))
   260	    _hash_array("audio_features", getattr(request, "audio_features", None))
   261	    _hash_array("audio_features_mask", getattr(request, "audio_features_mask", None))
   262	    if not media_sources:
   263	        # Fallback for callers that hand us preprocessed pixel tensors without
   264	        # source URLs/paths. Do not hash attention_mask: it changes with text
   265	        # history length and would make the same image miss across turns.
   266	        _hash_array("pixel_values", getattr(request, "pixel_values", None))
   267	        _hash_array(
   268	            "pixel_values_videos",
   269	            getattr(request, "pixel_values_videos", None)
   270	            if getattr(request, "pixel_values_videos", None) is not None
   271	            else getattr(request, "video_pixel_values", None),
   272	        )
   273
   274	    return {"mllm_media": hasher.hexdigest()}

    74	def _m3_vl_cached_prefix_covers_media_tokens(model: Any, request: Any) -> bool:
    75	    """Return True only when restored M3 state contains every media splice.
    76
    77	    A media-salted paged prefix is safe to replay without pixel tensors once
    78	    the cached boundary is strictly past the final image/video placeholder.
    79	    Partial hits that end inside the media span must take the original atomic
    80	    full-prompt vision forward instead.
    81	    """
    82	    cached_tokens = max(0, int(getattr(request, "cached_tokens", 0) or 0))
    83	    prompt_tokens = list(getattr(request, "prompt_token_ids", None) or [])
    84	    if cached_tokens <= 0 or not prompt_tokens:
    85	        return False
    86
    87	    config = getattr(model, "args", None) or getattr(model, "config", None)
    88	    media_token_ids: set[int] = set()
    89	    if getattr(request, "pixel_values", None) is not None:
    90	        media_token_ids.add(int(getattr(config, "image_token_index", 200025)))
    91	    if getattr(request, "pixel_values_videos", None) is not None:
    92	        media_token_ids.add(int(getattr(config, "video_token_index", 200026)))
    93	    if not media_token_ids:
    94	        return False
    95
    96	    media_positions = [
    97	        index for index, token in enumerate(prompt_tokens) if int(token) in media_token_ids
    98	    ]
    99	    return bool(media_positions) and cached_tokens > max(media_positions)
